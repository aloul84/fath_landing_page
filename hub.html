PlayerController.cs Unity C# 
using Fath.Play.SDK;
using UnityEngine;

public class PlayerController : MonoBehaviour
{
    void OnEnable()
    {
        // Zero CV setup required; SDK dispatches ready motion events
        MotionAPI.On(Gesture.HandUp,    Jump);
        MotionAPI.On(Gesture.HandRight, MoveRight);
        MotionAPI.On(Gesture.HandLeft,  MoveLeft);
    }

    void Jump() => body.AddForce(Vector2.up * 8.5f, ForceMode2D.Impulse);

    void OnLevelComplete(int score)
    {
        AchievementAPI.Unlock("FAST_REFLEXES");
        FathAnalytics.TrackScore(score);
    }
}
