فذ هب هو بوابة المطوّرين في منظومة فذ بلاي. ابنِ لعبتك بـ Unity 2D، واستخدم Motion API دون أن تلمس الكاميرا، ثم انشرها لآلاف العائلات العربية واحصل على حصّتك من المبيعات.

سجّل اهتمامك استكشف الـ SDK
PlayerController.cs 
using Fath.Play.SDK;

public class PlayerController : MonoBehaviour
{
    void OnEnable()
    {
        // No camera code — just motion events
        MotionAPI.On(Gesture.HandUp,    Jump);
        MotionAPI.On(Gesture.HandRight, MoveRight);
        MotionAPI.On(Gesture.HandLeft,  MoveLeft);
    }

    void Jump() => body.AddForce(Vector2.up * 8f);

    void OnLevelComplete(int score)
    {
        AchievementAPI.Unlock("first_win");
        SaveDataAPI.Set("best", score);
    }
}
